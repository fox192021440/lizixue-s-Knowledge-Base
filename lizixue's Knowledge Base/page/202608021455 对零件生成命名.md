Attribute VB_Name = "GenerateCode"

Option Explicit

  

'==========================================================================

' 宏功能：生成零件/装配体编码

' 编码格式：项目首字母缩写 + 日期(YYYYMMDD) + 三位流水号 + 零件/装配体名称

' 日期规则：检查保存文件夹内已有文件的日期，使用相同日期；没有则使用当天日期

' 流水号规则：零件和装配体分别独立编号，各自从最小可用号开始，避免重复

' 兼容性：SolidWorks 2021 及以上版本（使用晚期绑定，无需类型库引用）

' 使用方法：

'   方式一：直接打开零件文件 → 运行宏 → 重命名零件

'   方式二：直接打开装配体文件 → 运行宏 → 重命名装配体本身

'   方式三：在装配体中编辑零件（右键零件 → 编辑零件）→ 运行宏 → 重命名该零件

'   方式四：在总装配体中编辑子装配体（右键子装配体 → 编辑子装配体）

'          → 运行宏 → 重命名正在被编辑的子装配体

'==========================================================================

  

Dim swApp As Object

  

' 文档类型常量

Const DOC_PART = 1

Const DOC_ASSEMBLY = 2

Const DOC_DRAWING = 3

  

Sub main()

    On Error GoTo ErrorHandler

  

    Dim swModel As Object

    Dim docType As Long

    Dim swTargetModel As Object       ' 要重命名的目标模型

    Dim targetPath As String          ' 目标文件路径

    Dim folderPath As String          ' 文件夹路径

    Dim docName As String             ' 文件名（不含扩展名）

    Dim projectAbbr As String         ' 项目首字母缩写

    Dim useDate As String             ' 日期

    Dim serialNum As Long             ' 流水号

    Dim code As String                ' 生成的编码

    Dim targetDocType As Long         ' 目标文档类型

    Dim swAssy As Object              ' 装配体文档（如果在装配体内编辑）

    Dim isInContext As Boolean        ' 是否在装配体中编辑目标

  

    Set swApp = Application.SldWorks

    Set swModel = swApp.ActiveDoc

  

    ' 版本检测

    Dim revNum As String

    revNum = swApp.RevisionNumber

    ' RevisionNumber 返回格式如 "30.0.0"，主版本号对应：

    ' 30 = SW2022, 29 = SW2021, 28 = SW2020, 27 = SW2019 ...

    Dim mainVer As Long

    mainVer = CLng(Left(revNum, InStr(revNum, ".") - 1))

    If mainVer < 29 Then

        If MsgBox("检测到 SolidWorks 版本可能低于 2021（当前: " & revNum & "）。" & _

                  vbCrLf & "某些API可能不兼容，是否继续？", _

                  vbQuestion + vbYesNo, "版本兼容性提示") = vbNo Then

            Exit Sub

        End If

    End If

  

    ' 检查是否打开了文件

    If swModel Is Nothing Then

        MsgBox "请先打开一个零件或装配体文件！", vbExclamation, "提示"

        Exit Sub

    End If

  

    docType = swModel.GetType

  

    ' 根据文档类型确定重命名目标

    If docType = DOC_PART Then

        ' ====== 场景一：直接打开零件 ======

        Set swTargetModel = swModel

        targetPath = swModel.GetPathName

        targetDocType = DOC_PART

        Set swAssy = Nothing

        isInContext = False

    ElseIf docType = DOC_ASSEMBLY Then

        ' ====== 场景二/三/四：装配体 ======

        Set swAssy = swModel

  

        ' 获取正在编辑的模型文档

        Dim swEditModel As Object

        Set swEditModel = swAssy.GetEditTarget

  

        ' IsEditingSelf = True 表示装配体本身是编辑目标（没有编辑子装配体或零件）

        ' IsEditingSelf = False 表示正在编辑子装配体或零件

        Dim isEditingSelf As Boolean

        isEditingSelf = True

        On Error Resume Next

        isEditingSelf = swModel.IsEditingSelf

        On Error GoTo ErrorHandler

  

        If Not swEditModel Is Nothing And Not isEditingSelf Then

            ' 正在编辑子装配体或零件

            Dim editType As Long

            editType = swEditModel.GetType

  

            If editType = DOC_PART Then

                ' ====== 场景三：在装配体中编辑零件 ======

                Set swTargetModel = swEditModel

                targetPath = swTargetModel.GetPathName

                targetDocType = DOC_PART

                isInContext = True

            ElseIf editType = DOC_ASSEMBLY Then

                ' ====== 场景四：在总装配体中编辑子装配体 ======

                Set swTargetModel = swEditModel

                targetPath = swTargetModel.GetPathName

                targetDocType = DOC_ASSEMBLY

                isInContext = True

            Else

                ' 其他类型，回退到装配体本身

                Set swTargetModel = swModel

                targetPath = swModel.GetPathName

                targetDocType = DOC_ASSEMBLY

                isInContext = False

            End If

        Else

            ' ====== 场景二：装配体本身 ======

            Set swTargetModel = swModel

            targetPath = swModel.GetPathName

            targetDocType = DOC_ASSEMBLY

            isInContext = False

        End If

    Else

        MsgBox "工程图文件不支持此宏！", vbExclamation, "提示"

        Exit Sub

    End If

  

    ' 检查文件是否已保存

    If targetPath = "" Then

        MsgBox "文件未保存，请先保存文件后再运行此宏！", vbExclamation, "提示"

        Exit Sub

    End If

  

    ' 获取文件夹路径（含末尾反斜杠）

    folderPath = Left(targetPath, InStrRev(targetPath, "\"))

  

    ' 获取文件名（不含扩展名）

    docName = GetDocName(targetPath)

  

    ' 输入项目缩写

    projectAbbr = InputBox("请输入项目首字母缩写：", "项目缩写", "")

    If Trim(projectAbbr) = "" Then Exit Sub

    projectAbbr = Trim(projectAbbr)

  

    ' 去除文件名中已有的编码前缀（如果文件之前已编码过）

    docName = StripExistingCode(docName, projectAbbr)

  

    ' 如果去除编码后名称为空，提示用户

    If docName = "" Then

        MsgBox "无法获取有效的零件/装配体名称！", vbExclamation, "提示"

        Exit Sub

    End If

  

    ' 从文件夹内已有文件中获取日期（零件和装配体共同使用同一日期池）

    useDate = GetDateFromFolder(folderPath, projectAbbr)

  

    ' 获取下一个可用的流水号（按文档类型独立计数）

    serialNum = GetNextSerial(folderPath, projectAbbr, useDate, targetDocType)

  

    ' 生成编码

    code = projectAbbr & useDate & Format(serialNum, "000") & docName

  

    ' 显示编码并询问是否重命名

    Dim docTypeStr As String

    If targetDocType = DOC_PART Then

        docTypeStr = "零件"

    Else

        docTypeStr = "装配体"

    End If

  

    Dim userInput As String

    userInput = InputBox("生成的" & docTypeStr & "编码如下（可修改）：" & vbCrLf & vbCrLf & _

                         "格式：项目缩写 + 日期 + 流水号 + 名称" & vbCrLf & _

                         "类型：" & docTypeStr & vbCrLf & _

                         "点击确定 → 重命名文件" & vbCrLf & _

                         "点击取消 → 仅查看编码", _

                         "编码生成 - " & docTypeStr, code)

  

    If userInput <> "" Then

        If isInContext Then

            ' 在装配体中重命名正在编辑的目标（零件或子装配体）

            RenameInContextDoc swAssy, swTargetModel, targetPath, folderPath, userInput, targetDocType

        Else

            ' 重命名独立文件（零件或装配体）

            RenameFile swApp, swTargetModel, targetPath, folderPath, userInput, targetDocType

        End If

    End If

  

    Exit Sub

  

ErrorHandler:

    MsgBox "运行时错误 #" & Err.Number & ":" & vbCrLf & _

           Err.Description & vbCrLf & vbCrLf & _

           "请检查文件是否被占用或路径是否正确。", _

           vbCritical, "宏运行错误"

End Sub

  

'==========================================================================

' 从文件路径中提取文件名（不含扩展名）

'==========================================================================

Function GetDocName(filePath As String) As String

    On Error Resume Next

    Dim fileName As String

    fileName = Mid(filePath, InStrRev(filePath, "\") + 1)

    If InStr(fileName, ".") > 0 Then

        GetDocName = Left(fileName, InStrRev(fileName, ".") - 1)

    Else

        GetDocName = fileName

    End If

End Function

  

'==========================================================================

' 去除文件名中已有的编码前缀

' 匹配模式：[项目缩写][8位日期][3位流水号]

' 例如：ABC20260806001齿轮 → 齿轮

'==========================================================================

Function StripExistingCode(docName As String, projectAbbr As String) As String

    On Error Resume Next

    Dim prefixLen As Long

    prefixLen = Len(projectAbbr)

  

    If Left(docName, prefixLen) = projectAbbr And Len(docName) >= prefixLen + 11 Then

        Dim datePart As String

        Dim serialPart As String

        datePart = Mid(docName, prefixLen + 1, 8)

        serialPart = Mid(docName, prefixLen + 9, 3)

  

        If IsNumeric(datePart) And IsNumeric(serialPart) Then

            StripExistingCode = Mid(docName, prefixLen + 12)

            Exit Function

        End If

    End If

  

    StripExistingCode = docName

End Function

  

'==========================================================================

' 从文件夹内已有文件中获取日期

' 同时扫描零件和装配体文件，提取日期部分

' 如果找到多个文件，使用第一个匹配的日期

' 如果没有找到已有日期，使用当天日期

'==========================================================================

Function GetDateFromFolder(folderPath As String, projectAbbr As String) As String

    On Error Resume Next

    Dim fileName As String

    Dim foundDate As String

  

    foundDate = ""

    fileName = Dir(folderPath & projectAbbr & "*.sld*")

  

    Do While fileName <> ""

        If Len(fileName) >= Len(projectAbbr) + 8 Then

            Dim datePart As String

            datePart = Mid(fileName, Len(projectAbbr) + 1, 8)

            If IsNumeric(datePart) And Len(datePart) = 8 Then

                foundDate = datePart

                Exit Do

            End If

        End If

        fileName = Dir

    Loop

  

    If foundDate = "" Then

        foundDate = Format(Date, "yyyymmdd")

    End If

  

    GetDateFromFolder = foundDate

End Function

  

'==========================================================================

' 获取下一个可用的流水号（按文档类型独立计数）

' 零件只扫描 .sldprt 文件，装配体只扫描 .sldasm 文件

' 从小到大查找第一个未使用的号码

' 例如：零件已有 001, 002, 004 → 返回 003

'       装配体已有 001, 003 → 返回 002

'==========================================================================

Function GetNextSerial(folderPath As String, projectAbbr As String, _

                       useDate As String, docType As Long) As Long

    On Error Resume Next

    Dim fileName As String

    Dim prefix As String

    Dim serials() As Long

    Dim count As Long

    Dim i As Long

    Dim searchExt As String

  

    count = 0

    ReDim serials(0)

  

    ' 根据文档类型确定搜索扩展名

    If docType = DOC_PART Then

        searchExt = "*.sldprt"

    Else

        searchExt = "*.sldasm"

    End If

  

    prefix = projectAbbr & useDate

    fileName = Dir(folderPath & prefix & "*" & searchExt)

  

    ' 收集所有已使用的流水号

    Do While fileName <> ""

        If Len(fileName) >= Len(prefix) + 3 Then

            Dim serialStr As String

            serialStr = Mid(fileName, Len(prefix) + 1, 3)

            If IsNumeric(serialStr) Then

                ReDim Preserve serials(count)

                serials(count) = CLng(serialStr)

                count = count + 1

            End If

        End If

        fileName = Dir

    Loop

  

    ' 从1开始查找第一个未使用的流水号

    Dim testNum As Long

    Dim found As Boolean

  

    For testNum = 1 To 999

        found = False

        For i = 0 To count - 1

            If serials(i) = testNum Then

                found = True

                Exit For

            End If

        Next i

        If Not found Then

            GetNextSerial = testNum

            Exit Function

        End If

    Next testNum

  

    ' 如果1-999都被使用，返回1（理论上不太可能）

    GetNextSerial = 1

End Function

  

'==========================================================================

' 重命名独立文件（零件或装配体）

' 使用 OpenDoc6 替代 OpenDoc 以获得更好的错误诊断

'==========================================================================

Sub RenameFile(swApp As Object, swModel As Object, filePath As String, _

               folderPath As String, code As String, docType As Long)

    On Error GoTo RenameError

  

    Dim ext As String

    Dim newPath As String

    Dim docTitle As String

    Dim errors As Long

    Dim warnings As Long

  

    ext = Mid(filePath, InStrRev(filePath, "."))

    newPath = folderPath & code & ext

    docTitle = Mid(filePath, InStrRev(filePath, "\") + 1)

  

    ' 检查是否同名

    If LCase(filePath) = LCase(newPath) Then

        MsgBox "文件已经使用此编码命名！", vbInformation, "提示"

        Exit Sub

    End If

  

    ' 检查目标文件是否已存在

    If Dir(newPath) <> "" Then

        MsgBox "目标文件已存在：" & vbCrLf & newPath, vbExclamation, "错误"

        Exit Sub

    End If

  

    ' 关闭当前文档

    swApp.CloseDoc docTitle

  

    ' 重命名文件

    Name filePath As newPath

  

    ' 使用 OpenDoc6 重新打开（SW2007+ 可用，比 OpenDoc 更可靠）

    Dim swNewModel As Object

    Set swNewModel = swApp.OpenDoc6(newPath, docType, 0, "", errors, warnings)

  

    If swNewModel Is Nothing Then

        MsgBox "文件已重命名，但重新打开失败！" & vbCrLf & _

               "错误代码: " & errors & "，警告代码: " & warnings & vbCrLf & _

               "请手动打开：" & vbCrLf & newPath, vbExclamation, "提示"

    Else

        Set swModel = swNewModel

        MsgBox "文件已成功重命名为：" & vbCrLf & code & ext, vbInformation, "完成"

    End If

  

    Exit Sub

  

RenameError:

    MsgBox "重命名文件时发生错误 #" & Err.Number & ":" & vbCrLf & _

           Err.Description & vbCrLf & vbCrLf & _

           "原文件: " & filePath & vbCrLf & _

           "新文件: " & newPath, _

           vbCritical, "重命名错误"

End Sub

  

'==========================================================================

' 在装配体中重命名正在编辑的目标文档（零件或子装配体）

' SW2021 兼容版：优先使用 RenameDocument API，失败时回退到文件系统方式

' 注意：GetEditTargetComponent 和 RenameDocument 对零件和子装配体都适用

'==========================================================================

Sub RenameInContextDoc(swAssy As Object, swTargetModel As Object, _

                       targetPath As String, folderPath As String, _

                       code As String, targetDocType As Long)

    On Error GoTo RenameError

  

    Dim ext As String

    Dim newPath As String

    Dim errors As Long

    Dim warnings As Long

    Dim docTypeStr As String

  

    ext = Mid(targetPath, InStrRev(targetPath, "."))

    newPath = folderPath & code & ext

  

    If targetDocType = DOC_PART Then

        docTypeStr = "零件"

    Else

        docTypeStr = "子装配体"

    End If

  

    ' 检查是否同名

    If LCase(targetPath) = LCase(newPath) Then

        MsgBox "文件已经使用此编码命名！", vbInformation, "提示"

        Exit Sub

    End If

  

    ' 检查目标文件是否已存在

    If Dir(newPath) <> "" Then

        MsgBox "目标文件已存在：" & vbCrLf & newPath, vbExclamation, "错误"

        Exit Sub

    End If

  

    ' ====== 第一优先级：使用 RenameDocument API ======

    ' 这是 SW 官方推荐的方式，会自动更新装配体引用

    ' 对零件和子装配体均适用

    Dim swComp As Object

    Set swComp = swAssy.GetEditTargetComponent

  

    If swComp Is Nothing Then

        ' 无法获取组件，回退到文件系统方式

        GoTo FallbackRename

    End If

  

    ' 退出编辑模式，返回装配体编辑状态

    swAssy.EditAssembly

  

    ' 选择组件（RenameDocument 需要先选中组件）

    Dim selOK As Boolean

    selOK = swComp.Select2(False, 0)

  

    If Not selOK Then

        GoTo FallbackRename

    End If

  

    ' 使用 RenameDocument 重命名

    Dim swModelDocExt As Object

    Set swModelDocExt = swAssy.Extension

  

    errors = swModelDocExt.RenameDocument(code)

  

    If errors <> 0 Then

        ' RenameDocument 失败，回退到文件系统方式

        GoTo FallbackRename

    End If

  

    ' 重建装配体

    On Error Resume Next

    swAssy.EditRebuild3

    On Error GoTo RenameError

  

    ' 保存装配体及引用文件

    ' swSaveAsOptions_Silent = 1, swSaveAsOptions_SaveReferenced = 4

    Dim saveOK As Boolean

    saveOK = swAssy.Save3(5, errors, warnings)

  

    If Not saveOK Then

        MsgBox docTypeStr & "已重命名为：" & vbCrLf & code & ext & vbCrLf & vbCrLf & _

               "但保存装配体时出现提示。" & vbCrLf & _

               "错误代码: " & errors & "，警告代码: " & warnings & vbCrLf & _

               "请手动保存装配体以完成重命名。", vbExclamation, "提示"

    Else

        MsgBox docTypeStr & "已成功重命名为：" & vbCrLf & code & ext & vbCrLf & vbCrLf & _

               "装配体引用已自动更新。", vbInformation, "完成"

    End If

  

    Exit Sub

  

    ' ====== 降级方案：使用文件系统重命名 ======

FallbackRename:

    On Error GoTo FallbackError

  

    ' 重新进入编辑模式以保持状态

    On Error Resume Next

    swAssy.EditPart2 True, False, warnings

    On Error GoTo FallbackError

  

    ' 关闭目标文档

    Dim docTitle As String

    docTitle = Mid(targetPath, InStrRev(targetPath, "\") + 1)

    swApp.CloseDoc docTitle

  

    ' 在文件系统中重命名

    Name targetPath As newPath

  

    ' 重新打开目标文档

    Dim swNewDoc As Object

    Set swNewDoc = swApp.OpenDoc6(newPath, targetDocType, 0, "", errors, warnings)

  

    If swNewDoc Is Nothing Then

        MsgBox docTypeStr & "已重命名为：" & vbCrLf & code & ext & vbCrLf & vbCrLf & _

               "但重新打开失败（错误代码: " & errors & "）。" & vbCrLf & _

               "请手动打开" & docTypeStr & "并检查装配体引用。", vbExclamation, "提示"

    Else

        ' 重新在装配体中编辑目标

        On Error Resume Next

        swAssy.EditPart2 True, False, warnings

        On Error GoTo FallbackError

  

        ' 重建并保存装配体

        On Error Resume Next

        swAssy.EditRebuild3

        swAssy.Save3 5, errors, warnings

        On Error GoTo FallbackError

  

        MsgBox docTypeStr & "已重命名为：" & vbCrLf & code & ext & vbCrLf & vbCrLf & _

               "（使用文件系统方式重命名，可能需要在装配体中手动更新引用）", _

               vbInformation, "完成"

    End If

  

    Exit Sub

  

FallbackError:

    MsgBox "降级重命名也失败了！错误 #" & Err.Number & ":" & vbCrLf & _

           Err.Description & vbCrLf & vbCrLf & _

           "原文件: " & targetPath & vbCrLf & _

           "新文件: " & newPath, _

           vbCritical, "重命名失败"

  

    Exit Sub

  

RenameError:

    MsgBox "重命名" & docTypeStr & "时发生错误 #" & Err.Number & ":" & vbCrLf & _

           Err.Description & vbCrLf & vbCrLf & _
