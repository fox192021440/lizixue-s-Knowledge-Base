```vba
Option Explicit

' ==================== 配置区域 ====================
' 流水号文件存放目录
Const SERIAL_DIR As String = "D:\SW_SerialNumbers\"
' 流水号位数（默认3位 = 001~999）
Const SERIAL_DIGITS As Integer = 3
' 自定义属性名称
Const CUSTOM_PROP_NAME As String = "PartNo"
' ==================================================

Dim swApp As SldWorks.SldWorks

Sub main()
    Set swApp = Application.SldWorks
    
    Dim swModel As SldWorks.ModelDoc2
    Set swModel = swApp.ActiveDoc
    
    ' 检查是否有打开的零件
    If swModel Is Nothing Then
        MsgBox "请先打开或新建一个零件文件！", vbExclamation, "提示"
        Exit Sub
    End If
    
    If swModel.GetType() <> swDocPART Then
        MsgBox "当前文件不是零件文件（.sldprt）！", vbExclamation, "提示"
        Exit Sub
    End If
    
    ' ========== 第1步：输入项目缩写 ==========
    Dim projectCode As String
    projectCode = InputBox("请输入项目缩写（如：PRJ、ABC）：", "步骤 1/2 - 项目代号", "")
    
    If projectCode = "" Then
        MsgBox "未输入项目缩写，操作已取消。", vbInformation, "取消"
        Exit Sub
    End If
    
    projectCode = UCase(Trim(projectCode))
    
    ' ========== 自动生成日期和流水号 ==========
    Dim dateStr As String
    dateStr = Format(Now, "yymmdd")
    
    EnsureDirectoryExists SERIAL_DIR
    
    Dim serialFilePath As String
    serialFilePath = SERIAL_DIR & projectCode & "_" & dateStr & ".txt"
    
    Dim lastNumber As Long
    lastNumber = ReadSerialNumber(serialFilePath)
    
    Dim newNumber As Long
    newNumber = lastNumber + 1
    
    Dim maxNumber As Long
    maxNumber = 10 ^ SERIAL_DIGITS - 1
    
    If newNumber > maxNumber Then
        MsgBox "今日流水号已用完！最大支持 " & maxNumber & " 个零件。" & vbCrLf & _
               "请增加流水号位数或明日再试。", vbCritical, "流水号超限"
        Exit Sub
    End If
    
    Dim formatString As String
    formatString = String(SERIAL_DIGITS, "0")
    
    Dim serialPart As String
    serialPart = Format(newNumber, formatString)
    
    ' 基础编号（不含零件名称）
    Dim basePartNo As String
    basePartNo = projectCode & dateStr & serialPart
    
    ' ========== 第2步：输入零件名称 ==========
    Dim partName As String
    partName = InputBox("基础编号已生成：" & basePartNo & vbCrLf & vbCrLf & _
                        "请输入零件名称（如：轴承座、齿轮轴）：", _
                        "步骤 2/2 - 零件名称", "")
    
    ' 清理零件名称（去除首尾空格）
    partName = Trim(partName)
    
    ' 检查非法字符（文件名不允许的字符）
    Dim invalidChars As String
    invalidChars = "\/:*?""<>|"
    
    Dim i As Integer
    For i = 1 To Len(invalidChars)
        If InStr(partName, Mid(invalidChars, i, 1)) > 0 Then
            MsgBox "零件名称包含非法字符！请勿使用以下字符：" & vbCrLf & _
                   invalidChars, vbExclamation, "输入错误"
            Exit Sub
        End If
    Next i
    
    ' ========== 组合最终零件名 ==========
    Dim newPartName As String
    
    If partName = "" Then
        ' 未输入零件名称，仅使用基础编号
        newPartName = basePartNo
    Else
        ' 组合：基础编号 + 零件名称
        newPartName = basePartNo & partName
    End If
    
    ' ========== 设置零件标题 ==========
    If False = swModel.SetTitle2(newPartName) Then
        MsgBox "设置零件标题失败！", vbCritical, "错误"
        Exit Sub
    End If
    
    ' ========== 更新自定义属性 ==========
    Dim cpm As SldWorks.CustomPropertyManager
    Set cpm = swModel.Extension.CustomPropertyManager("")
    
    Dim retval As Long
    retval = cpm.Add2(CUSTOM_PROP_NAME, swCustomInfoText, newPartName)
    
    If retval <> 0 Then
        cpm.Set CUSTOM_PROP_NAME, newPartName
    End If
    
    ' 同时写入分项属性（方便工程图分别调用）
    cpm.Add2 "ProjectCode", swCustomInfoText, projectCode
    cpm.Set "ProjectCode", projectCode
    
    cpm.Add2 "DateCode", swCustomInfoText, dateStr
    cpm.Set "DateCode", dateStr
    
    cpm.Add2 "SerialNo", swCustomInfoText, serialPart
    cpm.Set "SerialNo", serialPart
    
    If partName <> "" Then
        cpm.Add2 "PartName", swCustomInfoText, partName
        cpm.Set "PartName", partName
    End If
    
    ' ========== 保存流水号 ==========
    StoreSerialNumber serialFilePath, newNumber
    
    ' ========== 完成提示 ==========
    MsgBox "零件命名成功！" & vbCrLf & vbCrLf & _
           "├─ 项目缩写：" & projectCode & vbCrLf & _
           "├─ 日期：" & dateStr & vbCrLf & _
           "├─ 流水号：" & serialPart & vbCrLf & _
           "├─ 零件名称：" & IIf(partName = "", "(未填写)", partName) & vbCrLf & _
           "└─ 完整名称：" & newPartName & vbCrLf & vbCrLf & _
           "自定义属性 [PartNo] 已同步更新。", vbInformation, "完成"
    
End Sub

' 读取流水号文件，如果不存在返回 0
Function ReadSerialNumber(filePath As String) As Long
    On Error GoTo FileNotFound
    
    Dim fileNo As Integer
    fileNo = FreeFile
    
    Dim numberStr As String
    
    Open filePath For Input As #fileNo
    Line Input #fileNo, numberStr
    Close #fileNo
    
    ReadSerialNumber = CLng(numberStr)
    Exit Function
    
FileNotFound:
    ReadSerialNumber = 0
End Function

' 保存流水号到文件
Sub StoreSerialNumber(filePath As String, number As Long)
    Dim fileNo As Integer
    fileNo = FreeFile
    
    Open filePath For Output As #fileNo
    Print #fileNo, CStr(number)
    Close #fileNo
End Sub

' 确保目录存在，不存在则创建
Sub EnsureDirectoryExists(dirPath As String)
    On Error Resume Next
    If Dir(dirPath, vbDirectory) = "" Then
        MkDir dirPath
    End If
    On Error GoTo 0
End Sub
```
